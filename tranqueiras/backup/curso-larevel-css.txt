a pasta pode estas dentro de resource mais com essas configurações abaixo e rode npm rum build para salvar as configurações 


configurações do blade

<!DOCTYPE html>
<html lang="{{ str_replace('_', '-', app()->getLocale()) }}">
<head>
    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1">
    <title>{{ config('app.name', 'Laravel') }}</title>

    <!-- Fonts -->
    <link rel="preconnect" href="https://fonts.bunny.net">
    <link href="https://fonts.bunny.net/css?family=instrument-sans:400,500,600" rel="stylesheet" />

    <!-- CSS -->
    @vite(['resources/css/app.css', 'resources/css/styles.css'])
    @vite(['resources/css/app.css', 'resources/css/estilo.css'])
</head>






configurações do vite

import { defineConfig } from 'vite';
import laravel from 'laravel-vite-plugin';

export default defineConfig({
    plugins: [
        laravel({
            input: ['resources/css/app.css', 'resources/css/styles.css', 'resources/css/estilo.css'],
            refresh: true,
        }),
    ],
});
